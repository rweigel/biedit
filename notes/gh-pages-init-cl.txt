# https://coderwall.com/p/mnwcog/create-new-github-repo-from-command-line

repo_name=$1
test -z $repo_name && echo "Repo name required." 1>&2 && exit 1
user_name=$2
test -z $user_name && echo "Repo name required." 1>&2 && exit 1

curl -u '$user_name' https://api.github.com/user/repos -d "{\"name\":\"$repo_name\"}"
git create $repo_name
git init
git remote add origin "https://github.com/$user_name/$repo_name.git"


# http://lea.verou.me/2011/10/easily-keep-gh-pages-in-sync-with-master/

git add .
git status 					# see what changes are to be commited
git commit -m 'Some descriptive commit message'
git push origin master

git checkout -b gh-pages  	# create branch
git checkout gh-pages 		# go to the gh-pages branch
git rebase master 			# bring gh-pages up to date with master
git push origin gh-pages 	# commit the changes
git checkout master 		# return to the master branch
